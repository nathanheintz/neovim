\newenvironment{coverpage}
  {\clearpage % we want a new page
  \thispagestyle{empty}% no header and footer
   \vspace*{\stretch{.6}}% some space at the top
   \flushright          % Right orientation
  }
  {\par % end the paragraph
   \vspace{\stretch{4}} % space at bottom is four times that at the top
  \clearpage           % finish off the page
  }

\newcounter{preamble}

% Define preamble command - similar to chapter but with italic TOC entry
\newcommand{\preamble}[1]{%
  \refstepcounter{preamble}%
  \cleardoublepage% Clear any previous content and start on odd page
  \chapter*{#1}% Creates unnumbered chapter
  \addcontentsline{toc}{preamble}{#1}% Add to TOC with preamble formatting
  \markboth{\textsc{\MakeLowercase{#1}}}{}% Update running header to show preamble in small caps
}

% Format preamble entries in TOC - italic, sentence case, no dots/page numbers
